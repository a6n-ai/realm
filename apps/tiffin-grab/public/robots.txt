# Public marketing pages are crawlable; the signed-in app and its APIs are not.
# Rules match by prefix, so the customer area uses /me$ and /me/ — a bare /me
# would also block /menu.
# Content Signals (contentsignals.org): search engines, AI answers that cite us, and AI training may all
# use the public pages. Being found and quoted is the point of them; flip ai-train to no to opt out of training.
User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=yes
Allow: /
Disallow: /api/
Disallow: /dashboard
Disallow: /me$
Disallow: /me/
Disallow: /checkout
Disallow: /activate
Disallow: /lock
